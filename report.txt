# Official documented RRC layout:
# AA 55 | Function | Length | Data[Length] | CRC
function Parse-RRC([byte[]]$Data) {
    $packets = New-Object System.Collections.Generic.List[object]

    for ($i=0; $i -lt $Data.Length-4; $i++) {
        if ($Data[$i] -ne 0xAA -or $Data[$i+1] -ne 0x55) { continue }

        $func = [int]$Data[$i+2]
        $len = [int]$Data[$i+3]
        $total = 5 + $len

        if ($len -gt 250) { continue }
        if ($i + $total -gt $Data.Length) { continue }

        $payload = [byte[]]@()
        if ($len -gt 0) {
            $payload = [byte[]]$Data[($i+4)..($i+3+$len)]
        }

        $crc = [int]$Data[$i+4+$len]

        [void]$packets.Add([pscustomobject]@{
            Offset = $i
            Function = $func
            Length = $len
            Payload = $payload
            CRC = $crc
            Raw = [byte[]]$Data[$i..($i+$total-1)]
        })

        $i += $total-1
    }

    @($packets)
}

function CRC8-07([byte[]]$Data) {
    $crc = 0
    foreach ($b in $Data) {
        $crc = ($crc -bxor [int]$b) -band 255
        for ($k=0; $k -lt 8; $k++) {
            if (($crc -band 0x80) -ne 0) {
                $crc = ((($crc -shl 1) -bxor 0x07) -band 255)
            } else {
                $crc = (($crc -shl 1) -band 255)
            }
        }
    }
    $crc
}

function RRC-CRCInput($p) {

}

    }
}

    foreach ($f in $Frames) {
        if ($f.Bytes.Length -gt $maxLen) { $maxLen = $f.Bytes.Length }
    }

    $rows = New-Object System.Collections.Generic.List[object]

    for ($pos=0; $pos -lt $maxLen; $pos++) {
        $vals = @()

        foreach ($f in $Frames) {
            if ($pos -lt $f.Bytes.Length) { $vals += [int]$f.Bytes[$pos] }
        }

        if ($vals.Count -lt 3) { continue }

        $distinct = @($vals | Sort-Object -Unique).Count
        $min = ($vals | Measure-Object -Minimum).Minimum
        $max = ($vals | Measure-Object -Maximum).Maximum

        $le = @()
        $be = @()

        foreach ($f in $Frames) {
            if ($pos+1 -lt $f.Bytes.Length) {
                $le += ([int]$f.Bytes[$pos]) -bor (([int]$f.Bytes[$pos+1]) -shl 8)
                $be += (([int]$f.Bytes[$pos] -shl 8) -bor [int]$f.Bytes[$pos+1])
            }
        }

        $constant = ($distinct -eq 1)
        $constText = ""
        if ($constant) { $constText = H $vals[0] }

        [void]$rows.Add([pscustomobject]@{
            Pos = $pos
            Samples = $vals.Count
            Distinct = $distinct
            Min = $min
            Max = $max
            Constant = $constant
            ConstValue = $constText
            U16LE_Min = ($le | Measure-Object -Minimum).Minimum
            U16LE_Max = ($le | Measure-Object -Maximum).Maximum
            U16BE_Min = ($be | Measure-Object -Minimum).Minimum
            U16BE_Max = ($be | Measure-Object -Maximum).Maximum
        })
    }

    @($rows)
}

function Compare-Frames([object[]]$Base,[object[]]$Action) {
    $n = [Math]::Min($Base.Count,$Action.Count)
    if ($n -lt 2) { return @() }

    $maxLen = 0
    for ($i=0; $i -lt $n; $i++) {
        $m = [Math]::Min($Base[$i].Bytes.Length,$Action[$i].Bytes.Length)
        if ($m -gt $maxLen) { $maxLen = $m }
    }

    $rows = New-Object System.Collections.Generic.List[object]

    for ($pos=0; $pos -lt $maxLen; $pos++) {
        $changed = 0
        $examples = New-Object System.Collections.Generic.List[string]

        for ($i=0; $i -lt $n; $i++) {
            $a = $Base[$i].Bytes
            $b = $Action[$i].Bytes

            if ($pos -lt $a.Length -and $pos -lt $b.Length -and $a[$pos] -ne $b[$pos]) {
                $changed++
                if ($examples.Count -lt 5) {
                    [void]$examples.Add(("{0:X2}->{1:X2}" -f $a[$pos],$b[$pos]))
                }
            }
        }

        if ($changed -gt 0) {
            [void]$rows.Add([pscustomobject]@{
                Position = $pos
                Changed = $changed
                Total = $n
                Fraction = [Math]::Round($changed/[double]$n,3)
                Examples = ($examples -join ", ")
            })
        }
    }

    @($rows | Sort-Object Fraction -Descending)
}

function H([int]$b) {
    "{0:X2}" -f ($b -band 255)
}

function HX([byte[]]$b) {
    if ($null -eq $b -or $b.Length -eq 0) { return "" }
    (($b | ForEach-Object { "{0:X2}" -f $_ }) -join " ")
}

function CountPair([byte[]]$Data,[int]$A,[int]$B) {
    $n = 0
    for ($i=0; $i -lt $Data.Length-1; $i++) {
        if ($Data[$i] -eq $A -and $Data[$i+1] -eq $B) { $n++ }
    }
    $n
}

function Get-DeviceName([string]$Com) {
    try {
        $escaped = [regex]::Escape($Com)
        $x = Get-CimInstance Win32_PnPEntity |
            Where-Object { $_.Name -and $_.Name -match "\($escaped\)$" } |
            Select-Object -First 1 -ExpandProperty Name
        if ($x) { return $x }
    } catch {}
    "unknown"
}

function Capture-Serial {
    param(
        [string]$Device,
        [int]$Rate,
        [string]$ParityName,
        [double]$Seconds
    )

    $sp = New-Object System.IO.Ports.SerialPort
    $sp.PortName = $Device
    $sp.BaudRate = $Rate
    $sp.DataBits = 8
    $sp.StopBits = [System.IO.Ports.StopBits]::One
    $sp.Handshake = [System.IO.Ports.Handshake]::None
    $sp.ReadTimeout = 50

    # Keep these off so we do not intentionally toggle/reset the target.
    $sp.DtrEnable = $false
    $sp.RtsEnable = $false

    if ($ParityName -eq "None") { $sp.Parity = [System.IO.Ports.Parity]::None }
    elseif ($ParityName -eq "Even") { $sp.Parity = [System.IO.Ports.Parity]::Even }
    elseif ($ParityName -eq "Odd") { $sp.Parity = [System.IO.Ports.Parity]::Odd }
    else { throw "Unsupported parity: $ParityName" }

    $buf = New-Object System.Collections.Generic.List[byte]

    try {
        $sp.Open()
        Start-Sleep -Milliseconds 100
        try { $sp.DiscardInBuffer() } catch {}

        $tmp = New-Object byte[] 8192
        $until = (Get-Date).AddSeconds($Seconds)

        while ((Get-Date) -lt $until) {
            try {
                $n = $sp.Read($tmp,0,$tmp.Length)
                if ($n -gt 0) {
                    for ($i=0; $i -lt $n; $i++) { [void]$buf.Add($tmp[$i]) }
                }
            } catch [System.TimeoutException] {}
        }

        return [byte[]]$buf.ToArray()
    }
    finally {
        if ($sp.IsOpen) { $sp.Close() }
        $sp.Dispose()
    }
}

# Official documented RRC layout:
# AA 55 | Function | Length | Data[Length] | CRC
function Parse-RRC([byte[]]$Data) {
    $packets = New-Object System.Collections.Generic.List[object]

    for ($i=0; $i -lt $Data.Length-4; $i++) {
        if ($Data[$i] -ne 0xAA -or $Data[$i+1] -ne 0x55) { continue }

        $func = [int]$Data[$i+2]
        $len = [int]$Data[$i+3]
        $total = 5 + $len

        if ($len -gt 250) { continue }
        if ($i + $total -gt $Data.Length) { continue }

        $payload = [byte[]]@()
        if ($len -gt 0) {
            $payload = [byte[]]$Data[($i+4)..($i+3+$len)]
        }

        $crc = [int]$Data[$i+4+$len]

        [void]$packets.Add([pscustomobject]@{
            Offset = $i
            Function = $func
            Length = $len
            Payload = $payload
            CRC = $crc
            Raw = [byte[]]$Data[$i..($i+$total-1)]
        })

        $i += $total-1
    }

    @($packets)
}

function CRC8-07([byte[]]$Data) {
    $crc = 0
    foreach ($b in $Data) {
        $crc = ($crc -bxor [int]$b) -band 255
        for ($k=0; $k -lt 8; $k++) {
            if (($crc -band 0x80) -ne 0) {
                $crc = ((($crc -shl 1) -bxor 0x07) -band 255)
            } else {
                $crc = (($crc -shl 1) -band 255)
            }
        }
    }
    $crc
}

function RRC-CRCInput($p) {
    $x = New-Object System.Collections.Generic.List[byte]
    [void]$x.Add([byte]$p.Function)
    [void]$x.Add([byte]$p.Length)
    foreach ($b in $p.Payload) { [void]$x.Add([byte]$b) }
    [byte[]]$x.ToArray()
}

function CRC-MatchRate([object[]]$Packets) {
    $matches = 0
    foreach ($p in $Packets) {
        if ((CRC8-07 (RRC-CRCInput $p)) -eq $p.CRC) { $matches++ }
    }
    [pscustomobject]@{
        Samples = $Packets.Count
        Matches = $matches
        Ratio = if ($Packets.Count -gt 0) {
            [Math]::Round($matches / [double]$Packets.Count,3)
        } else { 0 }
    }
}

function Top-Pairs([byte[]]$Data) {
    $d = @{}
    for ($i=0; $i -lt $Data.Length-1; $i++) {
        $k = "{0:X2} {1:X2}" -f $Data[$i],$Data[$i+1]
        if (-not $d.ContainsKey($k)) { $d[$k] = 0 }
        $d[$k]++
    }

    @(
        $d.GetEnumerator() |
        Sort-Object Value -Descending |
        Select-Object -First 50 |
        ForEach-Object {
            [pscustomobject]@{
                Pair = $_.Key
                Count = $_.Value
            }
        }
    )
}

function Stats([byte[]]$Data) {
    $u = @{}
    foreach ($b in $Data) { $u[[int]$b] = 1 }

    [pscustomobject]@{
        Bytes = $Data.Length
        Unique = $u.Count
        AA55 = CountPair $Data 0xAA 0x55
        CFC = CountPair $Data 0x0C 0xFC
        EFE = CountPair $Data 0x04 0xFE
    }
}

# Find repeated 2-byte header + 1-byte length candidates.
function Find-UnknownModels([byte[]]$Data) {
    $pairs = @(Top-Pairs $Data | Select-Object -First 30)
    $models = New-Object System.Collections.Generic.List[object]

    foreach ($pair in $pairs) {
        $parts = $pair.Pair.Split(" ")
        $h0 = [Convert]::ToInt32($parts[0],16)
        $h1 = [Convert]::ToInt32($parts[1],16)

        foreach ($lenPos in @(2,3,4,5)) {
            $hits = 0
            $valid = 0
            $next = 0
            $examples = New-Object System.Collections.Generic.List[int]

            for ($i=0; $i -lt $Data.Length-8; $i++) {
                if ($Data[$i] -ne $h0 -or $Data[$i+1] -ne $h1) { continue }
                $hits++

                $L = [int]$Data[$i+$lenPos]
                if ($L -gt 200) { continue }

                # Interpret length byte as payload length.
                $total = $lenPos + 2 + $L

                if ($i+$total -gt $Data.Length) { continue }
                $valid++

                if ($i+$total+1 -lt $Data.Length -and
                    $Data[$i+$total] -eq $h0 -and
                    $Data[$i+$total+1] -eq $h1) {
                    $next++
                }

                if ($examples.Count -lt 5) { [void]$examples.Add($L) }
            }

            if ($hits -ge 3) {
                $validRatio = [Math]::Round($valid/[double]$hits,3)
                if ($valid -gt 1) {
                    $continuity = [Math]::Round($next/[double]($valid-1),3)
                } else {
                    $continuity = 0
                }

                $score = [Math]::Round(
                    100 * (0.55*$validRatio + 0.45*$continuity),1
                )

                [void]$models.Add([pscustomobject]@{
                    Header = $pair.Pair
                    LenPos = $lenPos
                    Hits = $hits
                    Valid = $valid
                    NextHeader = $next
                    ValidRatio = $validRatio
                    Continuity = $continuity
                    Score = $score
                    ExampleLengths = ($examples -join ",")
                })
            }
        }
    }

    @(
        $models |
        Sort-Object Score -Descending |
        Select-Object -First 20
    )
}

function Parse-Unknown([byte[]]$Data,$Model) {
    $parts = $Model.Header.Split(" ")
    $h0 = [Convert]::ToInt32($parts[0],16)
    $h1 = [Convert]::ToInt32($parts[1],16)

    $frames = New-Object System.Collections.Generic.List[object]

    for ($i=0; $i -lt $Data.Length-8; $i++) {
        if ($Data[$i] -ne $h0 -or $Data[$i+1] -ne $h1) { continue }

        $L = [int]$Data[$i+$Model.LenPos]
        if ($L -gt 200) { continue }

        $total = $Model.LenPos + 2 + $L
        if ($i+$total -gt $Data.Length) { continue }

        [void]$frames.Add([pscustomobject]@{
            Offset = $i
            LengthField = $L
            Total = $total
            Bytes = [byte[]]$Data[$i..($i+$total-1)]
        })

        $i += $total-1
    }

    @($frames)
}

function Field-Stats([object[]]$Frames) {
    if ($Frames.Count -lt 3) { return @() }

    $maxLen = 0
    foreach ($f in $Frames) {
        if ($f.Bytes.Length -gt $maxLen) { $maxLen = $f.Bytes.Length }
    }

    $rows = New-Object System.Collections.Generic.List[object]

    for ($pos=0; $pos -lt $maxLen; $pos++) {
        $vals = @()

        foreach ($f in $Frames) {
            if ($pos -lt $f.Bytes.Length) { $vals += [int]$f.Bytes[$pos] }
        }

        if ($vals.Count -lt 3) { continue }

        $distinct = @($vals | Sort-Object -Unique).Count
        $min = ($vals | Measure-Object -Minimum).Minimum
        $max = ($vals | Measure-Object -Maximum).Maximum

        $le = @()
        $be = @()

        foreach ($f in $Frames) {
            if ($pos+1 -lt $f.Bytes.Length) {
                $le += ([int]$f.Bytes[$pos]) -bor (([int]$f.Bytes[$pos+1]) -shl 8)
                $be += (([int]$f.Bytes[$pos] -shl 8) -bor [int]$f.Bytes[$pos+1])
            }
        }

        $constant = ($distinct -eq 1)
        $constText = ""
        if ($constant) { $constText = H $vals[0] }

        [void]$rows.Add([pscustomobject]@{
            Pos = $pos
            Samples = $vals.Count
            Distinct = $distinct
            Min = $min
            Max = $max
            Constant = $constant
            ConstValue = $constText
            U16LE_Min = ($le | Measure-Object -Minimum).Minimum
            U16LE_Max = ($le | Measure-Object -Maximum).Maximum
            U16BE_Min = ($be | Measure-Object -Minimum).Minimum
            U16BE_Max = ($be | Measure-Object -Maximum).Maximum
        })
    }

    @($rows)
}

function Compare-Frames([object[]]$Base,[object[]]$Action) {
    $n = [Math]::Min($Base.Count,$Action.Count)
    if ($n -lt 2) { return @() }

    $maxLen = 0
    for ($i=0; $i -lt $n; $i++) {
        $m = [Math]::Min($Base[$i].Bytes.Length,$Action[$i].Bytes.Length)
        if ($m -gt $maxLen) { $maxLen = $m }
    }

    $rows = New-Object System.Collections.Generic.List[object]

    for ($pos=0; $pos -lt $maxLen; $pos++) {
        $changed = 0
        $examples = New-Object System.Collections.Generic.List[string]

        for ($i=0; $i -lt $n; $i++) {
            $a = $Base[$i].Bytes
            $b = $Action[$i].Bytes

            if ($pos -lt $a.Length -and $pos -lt $b.Length -and $a[$pos] -ne $b[$pos]) {
                $changed++
                if ($examples.Count -lt 5) {
                    [void]$examples.Add(("{0:X2}->{1:X2}" -f $a[$pos],$b[$pos]))
                }
            }
        }

        if ($changed -gt 0) {
            [void]$rows.Add([pscustomobject]@{
                Position = $pos
                Changed = $changed
                Total = $n
                Fraction = [Math]::Round($changed/[double]$n,3)
                Examples = ($examples -join ", ")
            })
        }
    }

    @($rows | Sort-Object Fraction -Descending)
}

try {
    Write-Host "============================================================" -ForegroundColor Cyan
    Write-Host " JETAUTO / HIWONDER UART PASSIVE RESEARCH" -ForegroundColor Cyan
    Write-Host "============================================================" -ForegroundColor Cyan
    Write-Host "Port      : $Port"
    Write-Host "Device    : $(Get-DeviceName $Port)"
    Write-Host "Bauds     : $($BaudPlan -join ', ')"
    Write-Host "Scan time : $ScanSeconds s/profile"
    Write-Host "Deep time : $DeepSeconds s"
    Write-Host "Mode      : PASSIVE (RX only)"
    Write-Host ""

    $ports = [System.IO.Ports.SerialPort]::GetPortNames()
    if ($ports -notcontains $Port) {
        throw "COM port $Port not found. Available: $($ports -join ', ')"
    }

    # --------------------------------------------------
    # PHASE 1: find the strongest serial profile
    # --------------------------------------------------
    Write-Host "PHASE 1 - SERIAL PROFILE SCAN" -ForegroundColor Cyan

    $profiles = New-Object System.Collections.Generic.List[object]

    foreach ($rate in $BaudPlan) {
        foreach ($par in @("None","Even","Odd")) {
            $parLetter = "N"
            if ($par -eq "Even") { $parLetter = "E" }
            elseif ($par -eq "Odd") { $parLetter = "O" }
            Write-Host ("Testing {0} / 8{1}1..." -f $rate, $parLetter)

            try {
                [byte[]]$cap = Capture-Serial $Port $rate $par $ScanSeconds
                $st = Stats $cap
                $rrc = @(Parse-RRC $cap)
                $crc = CRC-MatchRate $rrc

                $rawPath = Join-Path $OutDir ("scan_{0}_{1}.raw" -f $rate,$par)
                [IO.File]::WriteAllBytes($rawPath,$cap)

                $score = ($rrc.Count*1000) + ($crc.Matches*10000) + ($st.AA55*100) + ($st.Bytes/100.0)

                [void]$profiles.Add([pscustomobject]@{
                    Baud = $rate
                    Parity = $par
                    Bytes = $st.Bytes
                    AA55 = $st.AA55
                    CFC = $st.CFC
                    EFE = $st.EFE
                    RRC = $rrc.Count
                    CRC07 = $crc.Matches
                    Score = [Math]::Round($score,2)
                    File = $rawPath
                })

                Write-Host ("  bytes={0,6} AA55={1,4} RRC={2,4} CRC={3,4} 0CFC={4,4} 04FE={5,4} score={6}" -f `
                    $st.Bytes,$st.AA55,$rrc.Count,$crc.Matches,$st.CFC,$st.EFE,$score)
            }
            catch {
                Write-Host ("  FAILED: {0}" -f $_.Exception.Message) -ForegroundColor DarkYellow
            }
        }
    }

    if ($profiles.Count -eq 0) {
        throw "No serial profile could be opened."
    }

    $profiles |
        Sort-Object Score -Descending |
        Export-Csv -NoTypeInformation -Path (Join-Path $OutDir "profiles.csv")

    Write-Host ""
    Write-Host "TOP PROFILES" -ForegroundColor Cyan
    foreach ($p in $profiles | Sort-Object Score -Descending | Select-Object -First 15) {
        $pLetter = "N"
        if ($p.Parity -eq "Even") { $pLetter = "E" }
        elseif ($p.Parity -eq "Odd") { $pLetter = "O" }
        Write-Host ("  {0,8} 8{1}1 bytes={2,6} AA55={3,4} RRC={4,4} CRC={5,4} score={6}" -f `
            $p.Baud,$pLetter,$p.Bytes,$p.AA55,$p.RRC,$p.CRC07,$p.Score)
    }

    # Use explicit baud if requested, otherwise best structural profile.
    if ($Baud -gt 0) {
        $best = $profiles |
            Where-Object { $_.Baud -eq $Baud -and $_.Parity -eq "None" } |
            Sort-Object Score -Descending |
            Select-Object -First 1
    } else {
        $best = $profiles |
            Sort-Object Score -Descending |
            Select-Object -First 1
    }

    if ($null -eq $best) {
        throw "No usable profile selected."
    }

    # --------------------------------------------------
    # PHASE 2: deep capture
    # --------------------------------------------------
    Write-Host ""
    $bestLetter = "N"
    if ($best.Parity -eq "Even") { $bestLetter = "E" }
    elseif ($best.Parity -eq "Odd") { $bestLetter = "O" }
    Write-Host ("PHASE 2 - DEEP CAPTURE: {0} / 8{1}1" -f $best.Baud,$bestLetter) -ForegroundColor Cyan

    [byte[]]$deep = Capture-Serial $Port ([int]$best.Baud) $best.Parity $DeepSeconds
    $deepPath = Join-Path $OutDir "deep.raw"
    [IO.File]::WriteAllBytes($deepPath,$deep)

    Write-Host "Saved: $deepPath"
    Write-Host "Bytes: $($deep.Length)"

    if ($deep.Length -eq 0) {
        throw "Deep capture returned zero bytes."
    }

    $st = Stats $deep

    Write-Host ""
    Write-Host "STREAM COUNTS" -ForegroundColor Cyan
    Write-Host "  AA 55 : $($st.AA55)"
    Write-Host "  0C FC : $($st.CFC)"
    Write-Host "  04 FE : $($st.EFE)"

    Write-Host ""
    Write-Host "FIRST 128 BYTES" -ForegroundColor Cyan
    $show = [Math]::Min(128,$deep.Length)
    Write-Host (HX ([byte[]]$deep[0..($show-1)]))

    # --------------------------------------------------
    # PHASE 3: official RRC
    # --------------------------------------------------
    Write-Host ""
    Write-Host "PHASE 3 - OFFICIAL HIIWONDER RRC" -ForegroundColor Cyan
    Write-Host "Expected: AA 55 | Function | Length | Data | CRC"

    $rrc = @(Parse-RRC $deep)
    Write-Host "Complete RRC candidates: $($rrc.Count)"

    if ($rrc.Count -gt 0) {
        $crc = CRC-MatchRate $rrc
        Write-Host ("CRC-8/poly07 matches: {0}/{1} ({2})" -f `
            $crc.Matches,$crc.Samples,$crc.Ratio)

        $rows = foreach ($p in $rrc) {
            [pscustomobject]@{
                Offset = $p.Offset
                Function = ("0x{0:X2}" -f $p.Function)
                Length = $p.Length
                Payload = HX $p.Payload
                CRC = ("0x{0:X2}" -f $p.CRC)
                Raw = HX $p.Raw
            }
        }
        $rows | Export-Csv -NoTypeInformation -Path (Join-Path $OutDir "rrc_frames.csv")

        Write-Host ""
        Write-Host "Detected RRC functions:" -ForegroundColor Cyan
        foreach ($g in $rrc | Group-Object Function | Sort-Object Count -Descending) {
            Write-Host ("  Function 0x{0:X2} : {1} packet(s)" -f [int]$g.Name,$g.Count)
        }
    } else {
        Write-Host "No complete AA55 RRC packets."
    }

    # --------------------------------------------------
    # PHASE 4: unknown framing
    # --------------------------------------------------
    Write-Host ""
    Write-Host "PHASE 4 - UNKNOWN FRAME DISCOVERY" -ForegroundColor Cyan

    $models = @(Find-UnknownModels $deep)

    if ($models.Count -eq 0) {
        Write-Host "No repeated 2-byte-header + length model found."
        $bestUnknown = $null
    } else {
        foreach ($m in $models | Select-Object -First 15) {
            Write-Host ("  HEADER={0} LEN@{1} hits={2} valid={3} continuity={4} score={5} lengths={6}" -f `
                $m.Header,$m.LenPos,$m.Hits,$m.Valid,$m.Continuity,$m.Score,$m.ExampleLengths)
        }

        $bestUnknown = $models[0]

        if ($bestUnknown.Score -ge 65) {
            Write-Host ""
            Write-Host "BEST UNKNOWN MODEL" -ForegroundColor Green
            Write-Host "  Header : $($bestUnknown.Header)"
            Write-Host "  Length : byte position $($bestUnknown.LenPos)"
            Write-Host "  Score  : $($bestUnknown.Score)"

            $frames = @(Parse-Unknown $deep $bestUnknown)

            Write-Host "  Frames : $($frames.Count)"

            if ($frames.Count -ge 3) {
                $frameRows = foreach ($f in $frames) {
                    [pscustomobject]@{
                        Offset = $f.Offset
                        LengthField = $f.LengthField
                        Total = $f.Total
                        Hex = HX $f.Bytes
                    }
                }
                $frameRows | Export-Csv -NoTypeInformation -Path (Join-Path $OutDir "unknown_frames.csv")

                Write-Host ""
                Write-Host "DETECTED FRAMES" -ForegroundColor Cyan
                $n = 0
                foreach ($f in $frames | Select-Object -First 100) {
                    $n++
                    $frameHex = HX $f.Bytes
                    Write-Host ("FRAME {0,3}  offset={1,5} len={2,3}  {3}" -f `
                        $n,$f.Offset,$f.LengthField,$frameHex)
                }

                $fields = @(Field-Stats $frames)
                $fields | Export-Csv -NoTypeInformation -Path (Join-Path $OutDir "unknown_fields.csv")

                Write-Host ""
                Write-Host "BYTE POSITION STATISTICS" -ForegroundColor Cyan
                Write-Host "Pos Samples Distinct Min Max Const U16LE[min..max] U16BE[min..max]"

                foreach ($f in $fields) {
                    Write-Host ("{0,3} {1,7} {2,8} {3,3} {4,3} {5,-5} {6,15} {7,15}" -f `
                        $f.Pos,$f.Samples,$f.Distinct,$f.Min,$f.Max,$f.ConstValue, `
                        "$($f.U16LE_Min)..$($f.U16LE_Max)", `
                        "$($f.U16BE_Min)..$($f.U16BE_Max)")
                }
            }
        }
    }

    # --------------------------------------------------
    # PHASE 5: top pairs / old 0CFC hypothesis
    # --------------------------------------------------
    Write-Host ""
    Write-Host "PHASE 5 - TOP BYTE PAIRS" -ForegroundColor Cyan

    foreach ($p in Top-Pairs $deep | Select-Object -First 20) {
        Write-Host ("  {0} = {1}" -f $p.Pair,$p.Count)
    }

    if ($st.CFC -gt 0) {
        Write-Host ""
        if ($st.EFE -gt ($st.CFC*2)) {
            Write-Host "0C FC and 04 FE do NOT behave like a simple unique START/END pair in this capture." -ForegroundColor Yellow
        } else {
            Write-Host "0C FC / 04 FE occur, but this alone does NOT prove framing." -ForegroundColor Yellow
        }
    }

    # --------------------------------------------------
    # PHASE 6: controlled action experiment
    # --------------------------------------------------
    if (-not $NoActions) {
        Write-Host ""
        Write-Host "PHASE 6 - CONTROLLED ACTION EXPERIMENT" -ForegroundColor Cyan
        Write-Host "Passive only. This script never transmits."

        $run = Read-Host "Run one action experiment now? (Y/N)"

        if ($run -match '^[Yy]$') {
            $label = Read-Host "Action label (example: button_K1 / forward / left)"

            Write-Host ""
            Write-Host "Leave the robot idle."
            [void](Read-Host "Press ENTER to capture baseline")

            [byte[]]$base = Capture-Serial $Port ([int]$best.Baud) $best.Parity 3
            [IO.File]::WriteAllBytes((Join-Path $OutDir "action_baseline.raw"),$base)

            Write-Host ""
            Write-Host "Now press ENTER, then immediately perform: $label"
            [void](Read-Host "READY")

            [byte[]]$act = Capture-Serial $Port ([int]$best.Baud) $best.Parity 3
            [IO.File]::WriteAllBytes((Join-Path $OutDir "action.raw"),$act)

            if ($null -ne $bestUnknown -and $bestUnknown.Score -ge 65) {
                $bf = @(Parse-Unknown $base $bestUnknown)
                $af = @(Parse-Unknown $act $bestUnknown)
                $diff = @(Compare-Frames $bf $af)

                $diff | Export-Csv -NoTypeInformation `
                    -Path (Join-Path $OutDir "action_position_changes.csv")

                Write-Host ""
                Write-Host "POSITIONS CHANGED DURING ACTION: $label" -ForegroundColor Cyan
                foreach ($d in $diff | Select-Object -First 40) {
                    Write-Host ("  pos={0,3} changed={1,3}/{2,3} frac={3}  {4}" -f `
                        $d.Position,$d.Changed,$d.Total,$d.Fraction,$d.Examples)
                }
            } else {
                Write-Host "No high-confidence unknown frame model; raw baseline/action captures were saved."
            }
        }
    }

    Write-Host ""
    Write-Host "============================================================" -ForegroundColor Cyan
    Write-Host "DONE" -ForegroundColor Cyan
    Write-Host "Artifacts: $((Resolve-Path $OutDir).Path)"
    Write-Host "============================================================" -ForegroundColor Cyan
}
finally {
    try { Stop-Transcript | Out-Null } catch {}
}
